🐍 1. 線性迴歸 Python 實戰 
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score
import numpy as np

# 1. 準備讀書時數(X)與考分(y)
X = np.array([[1], [2], [4], [6], [8]])
y = np.array([2.5, 3.8, 5.2, 7.1, 8.9])

# 2. 建立模型並自動找出最佳直線 (OLS)
model = LinearRegression()
model.fit(X, y)

# 3. 印出斜率 w 與截距 b
print(f"最佳斜率 w = {model.coef_[0]:.2f}")
print(f"最佳截距 b = {model.intercept_:.2f}")
print(f"R2 解釋力 = {r2_score(y, model.predict(X)):.3f}")
🐍 2. 邏輯迴歸 Python 實戰 
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import confusion_matrix
import numpy as np

# 1. 模擬考分數(X) 與 是否錄取(y: 0或1)
X = np.array([[2], [3], [4.5], [6], [7.5], [9]])
y = np.array([0, 0, 0, 1, 1, 1])

# 2. 訓練 Sigmoid 分類模型
clf = LogisticRegression()
clf.fit(X, y)

# 3. 預測 5.5 分的錄取機率
prob = clf.predict_proba([[5.5]])[0, 1]
print(f"5.5 分錄取機率: {prob*100:.1f}%")
print("混淆矩陣:\n", confusion_matrix(y, clf.predict(X)))
🐍 3. KNN 近鄰分類 Python 實戰 
from sklearn.neighbors import KNeighborsClassifier
import numpy as np

# 1. 2D 座標特徵 (爆發力, 耐力) 與 隊伍標籤
X = np.array([[2,3], [3,2], [8,8], [7,9], [2,8]])
y = np.array(['藍隊', '藍隊', '紅隊', '紅隊', '黃隊'])

# 2. 設定找最近的 K=3 個鄰居，距離加權投票
knn = KNeighborsClassifier(
    n_neighbors=3,
    metric='euclidean',
    weights='distance'
)
knn.fit(X, y)

# 3. 放入新探針座標 [6, 7]
print("預測隊伍:", knn.predict([[6, 7]])[0])
